export async function runSync(
  accessToken: string,
  year: number
): Promise<SyncResult> {
  const now = new Date();
  const startDate = new Date(year, 0, 1); // Jan 1 of that year
  const endDate = year === now.getFullYear()
    ? now                          // current year: sync up to today
    : new Date(year, 11, 31, 23, 59, 59); // past year: sync full year

  logger.info(
    { year, start: startDate.toISOString(), end: endDate.toISOString() },
    "eBay sync: starting year fetch"
  );

  let fulfillmentOrders: FulfillmentOrder[] = [];
  try {

    const empty: SyncResult = {
      timestamp: new Date().toISOString(),
      transactions: [],
      counts: { sales: 0, fees: 0, expenses: 0, total: 0 },
    };
    lastResult = empty;
    return empty;
  }

2. Add syncedYears tracking to AppContext. This persists which years have been synced, alongside your existing AsyncStorage pattern:

javascript
// In AppContextValue interface:
syncedYears: number[];
markYearSynced: (year: number) => void;
isYearSyncable: (year: number) => boolean;
javascript
// In AppProvider, alongside your other useState calls:
const [syncedYears, setSyncedYears] = useState<number[]>([]);

// In the loading useEffect, alongside the other AsyncStorage.getItem calls:
const rawSyncedYears = await AsyncStorage.getItem('taxsquid_synced_years');
setSyncedYears(rawSyncedYears ? JSON.parse(rawSyncedYears) : []);

// New persistence effect:
useEffect(() => {
  if (!loaded) return;
  AsyncStorage.setItem('taxsquid_synced_years', JSON.stringify(syncedYears)).catch(() => {});
}, [syncedYears, loaded]);

// New callback:
const markYearSynced = useCallback((year: number) => {
  setSyncedYears((prev) => (prev.includes(year) ? prev : [...prev, year]));
}, []);

// Only current year and the immediately preceding year are ever syncable
const isYearSyncable = useCallback((year: number) => {
  const currentYear = new Date().getFullYear();
  return year === currentYear || year === currentYear - 1;
}, []);

Don't forget to add syncedYears, markYearSynced, and isYearSyncable to the value={{ ... }} object passed into AppContext.Provider.
